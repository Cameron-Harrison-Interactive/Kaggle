possible_paths = ['HI_Market_Brain.pkl', 'Agent/HI_Market_Brain.pkl']
ai_brain = None
for path in possible_paths:

        break

# --- Load Mutable DNA ---

# --- Core Agent Engine ---
def agent(observation, configuration):
    turn = observation.get('turn', 0)
    cash = observation.get('cash', 0)
    inventory = observation.get('inventory', {})
    livestock = observation.get('livestock', [])
    actions = []

    # RULE 1: DYNAMIC LIQUIDATION (Uses DNA)
    if turn >= dna["dump_turn"]:
        for item, qty in inventory.items():
            if qty > 0:
                actions.append(f"SELL_{item.upper()}_{qty}")
        for i in range(len(livestock)):
            actions.append(f"SELL_LIVESTOCK_{i}")
        actions.append("FIRE_ALL_LABOR")
        return actions if actions else ["DO_NOTHING"]

    # RULE 2: EMERGENCY FEED
    if len(livestock) > 0 and observation.get('animals_hungry', False):
        if inventory.get('wheat', 0) == 0 and cash > 25:
            return ["BUY_SEED WHEAT 1", "FEED_ANIMALS"]
        return ["FEED_ANIMALS"]

    # RULE 3: DYNAMIC CASH GUARD (Uses DNA)
    if cash <= dna["cash_guard"]:
        return ["DO_NOTHING"]

    # RULE 4: AI PREDICTION
    ai_prediction = 1
    if ai_brain and inventory.get('wheat', 0) > 0:
        try:
            day = observation.get('day', 1)
            hour = observation.get('hour', 12)
            price = observation.get('market_prices', {}).get('wheat', 25)
            features = np.array([[day, hour, price]], dtype=np.float32)
            ai_prediction = ai_brain.predict(features)[0]
        except Exception:
            ai_prediction = 1

    if inventory.get('wheat', 0) > 0:
        if ai_prediction == 1:
            return [f"SELL_WHEAT_{inventory['wheat']}"]
        else:
            pass 

    # RULE 5: DYNAMIC FARMING (Uses DNA)
    open_tiles = observation.get('open_tiles', 0)
    if open_tiles > 0 and cash > dna["plant_cash_threshold"]:
        return ["BUY_SEED WHEAT 1", "PLANT_WHEAT_1"]

    return ["DO_NOTHING"]

        cash = my_farm.get("money", 0)

        farmer_pos = my_farm.get("farmer", [0, 0])
        hands_pos = my_farm.get("hands", [])
        hires_today = my_farm.get("hires_today", 0)
        
        shed = private.get("shed", {})
        seeds = private.get("seeds", {})

        wheat_price = market_prices.get("WHEAT", 25)

        # PHASE 1: MARKET & LABOR RECRUITING
        # ----------------------------------
        if step >= global_dna["dump_turn"]:
            for item, qty in shed.items():
                if qty > 0 and len(market_queue) < 10:
                    market_queue.append(["SELL", item, qty])
        else:
            # Exploit cheap labor: Hire workers early in the day
            while hires_today < global_dna["target_hires"] and len(market_queue) < 10:
                market_queue.append(["HIRE"])
                hires_today += 1
                
            if shed.get("WHEAT", 0) > 0 and wheat_price >= global_dna["wheat_sell_price"]:
                if len(market_queue) < 10:
                    market_queue.append(["SELL", "WHEAT", shed["WHEAT"]])
            
            # Count living plants
            active_plants = sum(1 for row in tiles for t in row if isinstance(t, dict) and t.get("kind") == "PLANT")
            
            if (seeds.get("WHEAT", 0) + active_plants) < global_dna["max_plants"] and cash > global_dna["cash_guard"]:
                if len(market_queue) < 10:
                    market_queue.append(["BUY_SEED", "WHEAT", 1])

        action_dict["market"] = market_queue

        # ----------------------------------
        # PHASE 2: MULTI-AGENT TASK DISPATCHER
        # ----------------------------------
        unwatered = []
        harvestable = []
        empty = []
        weeds = []

                t = tiles[y][x]
                if t is None:
                    empty.append((x, y))
                elif isinstance(t, dict):
                    if t.get("kind") == "WEED":
                        weeds.append((x, y))
                    elif t.get("kind") == "PLANT":
                        crop_age = day - t.get("planted_day", day)
                        if crop_age >= 4:
                            harvestable.append((x, y))
                        elif not t.get("watered_today", False):
                            unwatered.append((x, y))

        # Combine main farmer and all hired hands into one workforce
        workers = [farmer_pos] + hands_pos
        worker_actions = []
        
        for wx, wy in workers:
            current_tile = tiles[wy][wx] if wy < len(tiles) and wx < len(tiles[0]) else None

                if crop_age >= 4 and (wx, wy) in harvestable:

            elif isinstance(current_tile, dict) and current_tile.get("kind") == "WEED" and (wx, wy) in weeds:
                act = ["DIG"]

            elif current_tile is None and seeds.get("WHEAT", 0) > 0 and (wx, wy) in empty:

                if target:

global_dna = {

}
initialized = False

def initialize_agent(player_idx):

    # Self-Play DNA Routing: Player 0 is the Challenger, Player 1 is the Reigning Champion
    if player_idx == 1:
        dna_path = 'Agent/champion_dna.json' if os.path.exists('Agent/champion_dna.json') else 'Agent/dna.json'
    else:
        dna_path = 'Agent/dna.json' if os.path.exists('Agent/dna.json') else 'dna.json'
        
    if os.path.exists(dna_path):
        try:
            with open(dna_path, 'r') as f:

            pass

def get_distance(x1, y1, x2, y2):
    return abs(x1 - x2) + abs(y1 - y2)

def move_towards(fx, fy, tx, ty):
    if fx < tx: return ["EAST"]
    if fx > tx: return ["WEST"]
    if fy < ty: return ["SOUTH"]
    if fy > ty: return ["NORTH"]
    return ["PASS"]

# ==========================================
# CORE AGENT ENGINE
# ==========================================
def agent(obs):
    try:
        player_idx = obs.get("player", 0)
        if not initialized:
            initialize_agent(player_idx)

        step = obs.get("step", 0)
        day = obs.get("day", 0) 
        farms = obs.get("farms", [])
        my_farm = farms[player_idx]
        private = obs.get("private", {})
        market_prices = obs.get("market", {}).get("prices", {})
        
        cash = my_farm.get("money", 0)
        tiles = my_farm.get("tiles", [])
        farmer_pos = my_farm.get("farmer", [0, 0])
        hands_pos = my_farm.get("hands", [])
        hires_today = my_farm.get("hires_today", 0)
        unlocked_quadrants = my_farm.get("unlocked_quadrants", ["NW"])
        
        shed = private.get("shed", {})
        seeds = private.get("seeds", {})
        
        action_dict = {"farmer": ["PASS"], "hands": [], "market": []}
        market_queue = []

        crop_type = global_dna.get("crop_choice", "WHEAT")
        base_seed_cost = 80 if crop_type == "MELON" else 10
        maturity_day = 12 if crop_type == "MELON" else 4
        market_base_price = 250 if crop_type == "MELON" else 25
        target_sell_price = market_base_price + global_dna["sell_price_offset"]
        current_product_price = market_prices.get(crop_type, market_base_price)

        # ----------------------------------
        # PHASE 1: MARKET, LAND EXPANSION & LABOR
        # ----------------------------------
        LAND_PRICES = [1000, 2000, 4000] #[cite: 1, 2]
        unlocked_extra = len(unlocked_quadrants) - 1

        if step >= global_dna["dump_turn"]:
            for item, qty in shed.items():
                if qty > 0 and len(market_queue) < 10:
                    market_queue.append(["SELL", item, qty])
        else:
            # Land Expansion Logic: Buy quadrants when we have enough cash safety buffer[cite: 2, 3]
            if unlocked_extra < global_dna["target_land_purchases"] and unlocked_extra < len(LAND_PRICES):
                land_cost = LAND_PRICES[unlocked_extra]
                if cash >= (land_cost + global_dna["cash_guard"] + 500) and len(market_queue) < 10:
                    market_queue.append(["BUY_LAND"])
                    cash -= land_cost

            # Hire workers
            while hires_today < global_dna["target_hires"] and len(market_queue) < 10:
                market_queue.append(["HIRE"])
                hires_today += 1
                
            # Sell Harvested Produce
            prod_in_shed = shed.get(crop_type, 0)
            if prod_in_shed > 0 and current_product_price >= target_sell_price:
                if len(market_queue) < 10:
                    market_queue.append(["SELL", crop_type, prod_in_shed])
            
            # Count active plants across the entire expanded board
            active_plants = sum(1 for row in tiles for t in row if isinstance(t, dict) and t.get("kind") == "PLANT")
            
            # Buy Seeds
            if (seeds.get(crop_type, 0) + active_plants) < global_dna["max_plants"] and cash > (global_dna["cash_guard"] + base_seed_cost):
                if len(market_queue) < 10:
                    market_queue.append(["BUY_SEED", crop_type, 1])

        action_dict["market"] = market_queue

        # ----------------------------------
        # PHASE 2: SWARM TASK DISPATCHER (UNLOCKED TILES ONLY)
        # ----------------------------------
        unwatered = []
        harvestable = []
        empty = []
        weeds = []
        
        for y in range(len(tiles)):
            for x in range(len(tiles[y])):
                t = tiles[y][x]
                if t == "LOCKED":
                    continue # Skip locked tiles as per game rules[cite: 3]
                elif t is None:
                    empty.append((x, y))
                elif isinstance(t, dict):
                    if t.get("kind") == "WEED":
                        weeds.append((x, y))
                    elif t.get("kind") == "PLANT":
                        crop_age = day - t.get("planted_day", day)
                        if crop_age >= maturity_day:
                            harvestable.append((x, y))
                        elif not t.get("watered_today", False):
                            unwatered.append((x, y))

        workers = [farmer_pos] + hands_pos
        worker_actions = []
        
        for wx, wy in workers:
            if wy >= len(tiles) or wx >= len(tiles[wy]):
                worker_actions.append(["PASS"])
                continue
                
            current_tile = tiles[wy][wx]
            act = ["PASS"]
            
            if current_tile == "LOCKED":
                # If a worker is on a locked tile, move them towards unlocked space
                worker_actions.append(["NORTH"])
                continue

            if isinstance(current_tile, dict) and current_tile.get("kind") == "PLANT":
                crop_age = day - current_tile.get("planted_day", day)
                if crop_age >= maturity_day and (wx, wy) in harvestable:
                    act = ["HARVEST"]
                    harvestable.remove((wx, wy))
                elif not current_tile.get("watered_today", False) and (wx, wy) in unwatered:
                    act = ["WATER"]
                    unwatered.remove((wx, wy))
            elif isinstance(current_tile, dict) and current_tile.get("kind") == "WEED" and (wx, wy) in weeds:
                act = ["DIG"]
                weeds.remove((wx, wy))
            elif current_tile is None and seeds.get(crop_type, 0) > 0 and (wx, wy) in empty:
                act = ["PLANT", crop_type]
                empty.remove((wx, wy))
                seeds[crop_type] -= 1 
            else:
                target = None
                if harvestable:
                    target = min(harvestable, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif unwatered:
                    target = min(unwatered, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif weeds:
                    target = min(weeds, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif empty and seeds.get(crop_type, 0) > 0:
                    target = min(empty, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                    
                if target:
                    act = move_towards(wx, wy, target[0], target[1])
            
            worker_actions.append(act)

        if len(worker_actions) > 0:
            action_dict["farmer"] = worker_actions[0]
        if len(worker_actions) > 1:
            action_dict["hands"] = worker_actions[1:]

        return action_dict
        
    except Exception as e:
        return {"farmer": ["PASS"], "hands": [], "market": []}

                worker_actions.append(["PASS"])
                continue

            act = ["PASS"]
            
            if current_tile == "LOCKED":
                worker_actions.append(["NORTH"])
                continue

            if isinstance(current_tile, dict) and current_tile.get("kind") == "PLANT":
                c_name = current_tile.get("crop")
                if c_name in CROP_CONFIGS:
                    cfg = CROP_CONFIGS[c_name]
                    crop_age = day - current_tile.get("planted_day", day)
                    yield_units = current_tile.get("yield_units", 0)
                    ready_to_harvest = (cfg["ongoing"] and yield_units > 0) or (not cfg["ongoing"] and crop_age >= cfg["maturity"])
                    
                    if ready_to_harvest and (wx, wy) in harvestable:
                        act = ["HARVEST"]
                        harvestable.remove((wx, wy))
                    elif not current_tile.get("watered_today", False) and (wx, wy) in unwatered:
                        act = ["WATER"]
                        unwatered.remove((wx, wy))
            elif isinstance(current_tile, dict) and current_tile.get("kind") == "WEED" and (wx, wy) in weeds:
                act = ["DIG"]
                weeds.remove((wx, wy))
            elif current_tile is None and (wx, wy) in empty:
                available_seed_type = next((s for s, count in seeds.items() if count > 0), None)
                if available_seed_type:
                    act = ["PLANT", available_seed_type]
                    empty.remove((wx, wy))
                    seeds[available_seed_type] -= 1
            else:
                target = None
                if harvestable:
                    target = min(harvestable, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif unwatered:
                    target = min(unwatered, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif weeds:
                    target = min(weeds, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                elif empty and sum(seeds.values()) > 0:
                    target = min(empty, key=lambda p: get_distance(wx, wy, p[0], p[1]))
                    
                if target:
                    act = move_towards(wx, wy, target[0], target[1])
            
            worker_actions.append(act)

        if len(worker_actions) > 0:
            action_dict["farmer"] = worker_actions[0]
        if len(worker_actions) > 1:
            action_dict["hands"] = worker_actions[1:]

        return action_dict
